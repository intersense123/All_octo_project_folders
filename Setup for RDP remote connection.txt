On docker-compose.yml add below lines:

services:
  weston:
    image: torizon/weston:$CT_TAG_WESTON
    restart: always
    privileged: true
    network_mode: host
    ipc: host

    environment:
      - ENABLE_RDP=1

    volumes:
      - /tmp:/tmp
      - /dev:/dev
      - /run/udev:/run/udev
      - /home/torizon/custom:/etc/xdg/weston-dev/
      - /home/torizon/custom/Loading.png:/home/torizon/custom/Loading.png

    cap_add:
      - CAP_SYS_TTY_CONFIG

    device_cgroup_rules:
      - 'c 4:* rmw'
      - 'c 13:* rmw'
      - 'c 226:* rmw'
      - 'c 10:223 rmw'
